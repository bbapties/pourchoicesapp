CREATE TABLE public.bottle_variants (
  id uuid NOT NULL DEFAULT uuid_generate_v4(),
  bottles_id uuid NOT NULL,
  frontimage_url text,
  backimage_url text,
  age text,
  proof numeric,
  batch text,
  release_year integer,
  store_pick_name text,
  verified boolean NOT NULL DEFAULT false,
  created_by uuid,
  created_at timestamp with time zone DEFAULT CURRENT_TIMESTAMP,
  updated_by uuid,
  updated_at timestamp with time zone DEFAULT CURRENT_TIMESTAMP,
  notes text,
  CONSTRAINT bottle_variants_pkey PRIMARY KEY (id),
  CONSTRAINT bottle_attr_bottles_id_fkey FOREIGN KEY (bottles_id) REFERENCES public.bottles(id),
  CONSTRAINT bottle_attr_updated_by_fkey FOREIGN KEY (updated_by) REFERENCES auth.users(id)
);
CREATE TABLE public.bottles (
  id uuid NOT NULL DEFAULT uuid_generate_v4(),
  name text NOT NULL,
  category text,
  verified boolean DEFAULT false,
  elo_global numeric DEFAULT 1500,
  created_at timestamp with time zone DEFAULT now(),
  updated_at timestamp with time zone DEFAULT now(),
  distillery text,
  style text,
  barcode text,
  created_by uuid,
  updated_by uuid,
  nose text,
  palate text,
  finish text,
  extras text,
  proof numeric,
  age text,
  volume text,
  frontimage_url text,
  backimage_url text,
  CONSTRAINT bottles_pkey PRIMARY KEY (id),
  CONSTRAINT bottles_created_by_fkey FOREIGN KEY (created_by) REFERENCES auth.users(id),
  CONSTRAINT bottles_updated_by_fkey FOREIGN KEY (updated_by) REFERENCES auth.users(id)
);
CREATE TABLE public.tasting_details (
  id uuid NOT NULL DEFAULT uuid_generate_v4(),
  tasting_session_id uuid NOT NULL,
  bottle_id uuid NOT NULL,
  notes jsonb,
  created_at timestamp with time zone DEFAULT now(),
  CONSTRAINT tasting_details_pkey PRIMARY KEY (id),
  CONSTRAINT tasting_details_tasting_session_id_fkey FOREIGN KEY (tasting_session_id) REFERENCES public.tasting_sessions(id),
  CONSTRAINT tasting_details_bottle_id_fkey FOREIGN KEY (bottle_id) REFERENCES public.bottles(id)
);
CREATE TABLE public.tasting_results (
  id uuid NOT NULL DEFAULT uuid_generate_v4(),
  tasting_session_id uuid NOT NULL,
  winner_bottle_id uuid NOT NULL,
  loser_bottle_id uuid NOT NULL,
  created_at timestamp with time zone DEFAULT now(),
  CONSTRAINT tasting_results_pkey PRIMARY KEY (id),
  CONSTRAINT tasting_results_tasting_session_id_fkey FOREIGN KEY (tasting_session_id) REFERENCES public.tasting_sessions(id),
  CONSTRAINT tasting_results_winner_bottle_id_fkey FOREIGN KEY (winner_bottle_id) REFERENCES public.bottles(id),
  CONSTRAINT tasting_results_loser_bottle_id_fkey FOREIGN KEY (loser_bottle_id) REFERENCES public.bottles(id)
);
CREATE TABLE public.tasting_sessions (
  id uuid NOT NULL DEFAULT uuid_generate_v4(),
  user_id uuid NOT NULL,
  name text,
  bottle_ids ARRAY NOT NULL DEFAULT '{}'::uuid[],
  is_blind boolean DEFAULT true,
  created_at timestamp with time zone DEFAULT now(),
  CONSTRAINT tasting_sessions_pkey PRIMARY KEY (id),
  CONSTRAINT tasting_sessions_user_id_fkey FOREIGN KEY (user_id) REFERENCES public.users(id)
);
CREATE TABLE public.user_bottles (
  user_id uuid NOT NULL,
  bottle_id uuid NOT NULL,
  rating_stars numeric DEFAULT NULL::numeric CHECK (rating_stars >= 0::numeric AND rating_stars <= 5::numeric),
  elo numeric DEFAULT 1500,
  created_at timestamp with time zone DEFAULT now(),
  updated_at timestamp with time zone DEFAULT now(),
  currently_owned boolean DEFAULT true,
  variant_id uuid,
  CONSTRAINT user_bottles_pkey PRIMARY KEY (user_id, bottle_id),
  CONSTRAINT user_bottles_user_id_fkey FOREIGN KEY (user_id) REFERENCES public.users(id),
  CONSTRAINT user_bottles_bottle_id_fkey FOREIGN KEY (bottle_id) REFERENCES public.bottles(id),
  CONSTRAINT user_bottles_variant_id_fkey FOREIGN KEY (variant_id) REFERENCES public.bottle_variants(id)
);
CREATE TABLE public.users (
  id uuid NOT NULL DEFAULT uuid_generate_v4(),
  username text NOT NULL UNIQUE,
  email text NOT NULL UNIQUE,
  collection ARRAY DEFAULT '{}'::uuid[],
  created_at timestamp with time zone DEFAULT now(),
  auth_id uuid,
  CONSTRAINT users_pkey PRIMARY KEY (id),
  CONSTRAINT users_auth_id_fkey FOREIGN KEY (auth_id) REFERENCES auth.users(id)
);
